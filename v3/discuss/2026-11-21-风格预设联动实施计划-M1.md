/**
 * 风格→预设匹配纯函数（M1 抽取自 StylePresetSection，设计共识 D2/D3）：
 * - 关键词构造：题材 + 基调拆词 + 风格中文标签拆词
 * - 匹配：tag 与预设名称双向子串包含（「欢乐」命中名称「番茄欢乐」）
 * - 理由：命中的 tag/名称列表（确认弹窗展示「命中：××」）
 * - 分组：通用（常驻）/ 匹配推荐 / 其余
 */
import { WRITING_STYLE_LABELS, type WritingStyle } from '@shared/work';

export interface PresetLite {
  id: string;
  name: string;
  tags?: string[];
}

/** 预设 tag/名称 ↔ 书风格关键词 匹配（双向包含，如 tag「治愈」匹配基调「温情治愈」） */
export function tagMatches(tag: string, keywords: string[]): boolean {
  return keywords.some((k) => k && (tag.includes(k) || k.includes(tag)));
}

/** 书风格关键词：题材 + 各基调按「·」/空白拆词 + 风格中文标签拆词（过滤空串与「—」） */
export function buildStyleKeywords(
  subGenre: string,
  storyTones: string[],
  writingStyle: WritingStyle,
): string[] {
  return [subGenre, ...storyTones, WRITING_STYLE_LABELS[writingStyle]]
    .flatMap((k) => (k ?? '').split(/[·\s、]/))
    .map((k) => k.trim())
    .filter((k) => k && k !== '—');
}

/**
 * 匹配理由（D3：确认弹窗展示推荐依据）：
 * - tag 命中（「通用」除外——通用走常驻分组不算推荐理由）
 * - 名称命中（前缀「名称:」）
 */
export function matchReasons(name: string, tags: string[], keywords: string[]): string[] {
  const reasons: string[] = [];
  for (const tag of tags) {
    if (tag !== '通用' && tagMatches(tag, keywords)) reasons.push(tag);
  }
  if (tagMatches(name, keywords)) reasons.push(`名称:${name}`);
  return [...new Set(reasons)];
}

/** 预设三分组（D3 确认弹窗结构） */
export interface PresetGroups {
  general: PresetLite[];
  matched: PresetLite[];
  others: PresetLite[];
}

export function groupPresets(presets: PresetLite[], keywords: string[]): PresetGroups {
  const general: PresetLite[] = [];
  const matched: PresetLite[] = [];
  const others: PresetLite[] = [];
  for (const p of presets) {
    const tags = p.tags ?? [];
    if (tags.includes('通用')) general.push(p);
    else if (matchReasons(p.name, tags, keywords).length > 0) matched.push(p);
    else others.push(p);
  }
  return { general, matched, others };
}

```tsx
/**
 * 风格预设确认弹窗（设计共识 D3）：非通用预设生效的唯一入口——
 * 三分组（通用常驻默认勾选 / 匹配推荐预勾选+理由 / 其余可手动补选），
 * 弹窗内勾选可调，点「确认」回传最终勾选列表（由父组件写入 settings.stylePresetIds）。
 */
import { useState } from 'react';

import { groupPresets, type PresetLite } from './stylePresetMatch';

interface Props {
  presets: PresetLite[];
  /** 进入弹窗时的勾选草稿 */
  draft: string[];
  keywords: string[];
  onConfirm: (picked: string[]) => void;
  onClose: () => void;
}

export default function StylePresetConfirmDialog({ presets, draft, keywords, onConfirm, onClose }: Props) {
  const [picked, setPicked] = useState<string[]>(draft);
  const { general, matched, others } = groupPresets(presets, keywords);

  const toggle = (id: string): void =>
    setPicked((prev) => (prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id]));

  const row = (p: PresetLite, reason?: string) => (
    <label key={p.id} className="flex min-h-8 cursor-pointer items-center gap-2 rounded px-1.5 text-xs hover:bg-[hsl(var(--bg))]">
      <input
        type="checkbox"
        checked={picked.includes(p.id)}
        data-preset-id={p.id}
        onChange={() => toggle(p.id)}
        className="h-3.5 w-3.5 accent-[hsl(var(--accent))]"
      />
      <span className="truncate">{p.name}</span>
      {(p.tags ?? []).length > 0 && (
        <span className="ml-auto shrink-0 text-[10px] text-[hsl(var(--fg-muted))]">{(p.tags ?? []).join(' / ')}</span>
      )}
      {reason && <span className="shrink-0 rounded bg-[hsl(var(--accent))]/10 px-1 text-[10px] text-[hsl(var(--accent))]">命中：{reason}</span>}
    </label>
  );

  return (
    <div
      className="fixed inset-0 z-[60] flex items-center justify-center bg-black/60 p-4"
      role="dialog"
      aria-modal="true"
      aria-label="确认风格预设"
      data-testid="preset-confirm-dialog"
      onClick={onClose}
    >
      <div
        className="max-h-[80vh] w-full max-w-md overflow-y-auto rounded-2xl border border-[hsl(var(--border-soft))] bg-[hsl(var(--surface))] p-5 shadow-2xl"
        onClick={(e) => e.stopPropagation()}
      >
        <h3 className="mb-1 text-base font-semibold">确认风格预设</h3>
        <p className="mb-3 text-xs text-[hsl(var(--fg-muted))]">
          通用预设常驻；匹配推荐按书风格关键词给出（含理由），非通用预设确认后才会随生成注入。
        </p>
        {(
          [
            ['general', '通用（常驻）', general, undefined],
            ['matched', '匹配推荐', matched, keywords],
            ['others', '其余预设', others, undefined],
          ] as const
        ).map(([key, title, list]) => (
          <div key={key} className="mb-3" data-group={key}>
            <p className="mb-1 text-xs font-medium text-[hsl(var(--fg-muted))]">{title}</p>
            {list.length === 0 && <p className="px-1.5 py-1 text-[11px] text-[hsl(var(--fg-muted))]">无</p>}
            {list.map((p) =>
              key === 'matched' ? row(p, matchReasonText(p, keywords)) : row(p),
            )}
          </div>
        ))}
        <div className="flex justify-end gap-3">
          <button type="button" onClick={onClose} className="h-9 rounded-lg px-4 text-sm text-[hsl(var(--fg-muted))]">
            取消
          </button>
          <button
            type="button"
            onClick={() => onConfirm(picked)}
            data-testid="preset-confirm-ok"
            className="h-9 rounded-lg bg-[hsl(var(--accent))] px-5 text-sm font-medium text-white"
          >
            确认应用
          </button>
        </div>
      </div>
    </div>
  );
}

import { matchReasons } from './stylePresetMatch';
function matchReasonText(p: PresetLite, keywords: string[]): string {
  return matchReasons(p.name, p.tags ?? [], keywords).join(' / ');
}
```

2. **推荐重跑**（设计共识 M1.2）：`useEffect` 依赖从 `[work.id]` 扩展为 `[work.id, presets, style, s.storyTones, s.subGenre]`。风格/基调变化时：`auto = presets 中通用 ∪ matchReasons 非空`，`added = auto − picked − dismissed`；`added` 非空则并入 `picked` **并同时 `setDirty(true)`**（否则「确认应用」按钮 disabled，确认链路走不通），并记入 `newRecs`（列表 UI 打「新」徽标，`data-testid="preset-new-rec"`）。
3. **dismissed 语义（D3：用户明确取消的预设不偷偷加回）**：`dismissedRef: Set<string>` 记录两类来源——① 主列表手动取消勾选的 id；② **确认弹窗 onConfirm 时 `draft − 确认结果` 的差集**（弹窗内取消的同样不得在保存后因 work 更新触发 effect 重跑而弹回）。dismissed 为会话内存态，不持久化。
4. **确认弹窗接入**：`apply()` 拆两步——点「确认应用（风格+预设）」→ 打开 `StylePresetConfirmDialog`（draft=picked）；弹窗 onConfirm → dismissed 合并（见要点 3）→ 原 PATCH 流程（`stylePresetIds: picked`、`writingStyle: style`、`customWritingStyle`）。弹窗渲染条件 state `confirmOpen`。
5. **未确认提示条**：`!(s.stylePresetIds?.length)` 时区块顶部显示 `<p data-testid="preset-hint">尚未确认风格预设：勾选后点「确认应用」生效（不确认则仅主预设生效）</p>`——新建书（Task 4 的 NewWorkDialog 跳转）落地即见。
6. **匹配徽标理由**：列表行「匹配」徽标 `title` 属性 = matchReasons 结果（hover 可见），徽标文案不变；主清单勾选行 checkbox 加 `data-preset-id={p.id}`（与弹窗一致，测试定位）。
